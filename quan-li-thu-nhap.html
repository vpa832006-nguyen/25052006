Phân tích lỗi và sửa đoạn mã JavaScript

Mã GốcCó lỗi 
const changeTheme = () => {
let body = document.querySelector("body");
if (body.style.background == "white") {
  body.style.background = "black";
  body.style.color = "white";
} else {
  body.style.background = "white"
  body.style.color = "black"
}
}
Mã Đã SửaHoàn chỉnh 
// Lỗi 1: Dùng == thay vì ===
//   == không kiểm tra kiểu dữ liệu, dễ gây so sánh sai.
// Lỗi 2: body.style.background trả về "" khi chưa set bằng JS
//   → Điều kiện if(...=="white") luôn false lúc đầu.
//   → Sửa: dùng biến isDark để theo dõi trạng thái.
// Lỗi 3 & 4: Thiếu dấu ; cuối 2 dòng trong else
//   → JS có ASI nhưng thiếu
let isDark = false;
const changeTheme = () => {
  const body = document.querySelector("body");
  if (!isDark) {
    body.style.background = "black";
    body.style.color = "white";
    isDark = true;
  } else {
    body.style.background = "white";  // ✅ thêm ;
    body.style.color = "black";      // ✅ thêm ;
    isDark = false;
  }
};
